# AI Chat Service — Batch 14A (skeleton) dependencies.
# Web framework + runtime only. The RAG/AI stack (qdrant-client,
# sentence-transformers, google-generativeai), and data clients (motor, redis)
# are introduced in their own batches to keep the skeleton image small.

fastapi==0.115.6
uvicorn[standard]==0.34.0
pydantic==2.10.4
pydantic-settings==2.7.1

# Data clients (readiness checks; data access wired in later batches)
redis==5.2.1
motor==3.6.0

# Vector DB client (collection bootstrap + readiness; vectors/search wired later)
qdrant-client==1.12.1

# Embedding model (lazy-loaded on first use). sentence-transformers pulls in
# torch unpinned, which resolves to the default PyPI wheel — a full
# CUDA-enabled build (~2GB of nvidia-* runtime libraries) that this
# CPU-only deployment never uses (see embedder.py: device is always "cpu").
# On Render's 512Mi free tier, that extra weight was enough to OOM-kill the
# container while loading the model at startup, before the port ever opened.
# Pinning the CPU-only wheel (same torch version, no CUDA runtime) keeps
# behavior identical while cutting both image size and import-time memory.
--extra-index-url https://download.pytorch.org/whl/cpu
torch==2.7.1+cpu
sentence-transformers==3.3.1

# Groq generation (lazy-loaded; requires GROQ_API_KEY at runtime).
# Official SDK — supports native async via AsyncGroq (OpenAI-compatible API).
groq>=0.11,<1.0

# HTTP client for downloading reel video files (Cloudinary URLs) ahead of
# Whisper transcription — see provider.generate_reel_captions. Previously
# only a transitive dependency (qdrant-client/fastapi's test client); pinned
# explicitly since this is now a direct import.
httpx>=0.27,<1.0
